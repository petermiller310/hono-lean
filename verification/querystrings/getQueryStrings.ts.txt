export const getQueryStrings = (url: string): string => {
  const queryIndex = url.indexOf('?', 8)
  if (queryIndex === -1) {
    return ''
  }

  const hashIndex = url.indexOf('#', 8)
  return hashIndex === -1
    ? url.slice(queryIndex)
    : queryIndex < hashIndex
      ? url.slice(queryIndex, hashIndex)
      : ''
}
